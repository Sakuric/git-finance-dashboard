   ```java
   // src/main/java/com/example/financedashboard/service/impl/UserServiceImpl.java
   package com.example.financedashboard.service.impl;
   
   import com.example.financedashboard.dto.UserRegisterDTO;
   import com.example.financedashboard.dto.UserLoginDTO;
   import com.example.financedashboard.entity.User;
   import com.example.financedashboard.mapper.UserMapper;
   import com.example.financedashboard.service.UserService;
   import com.example.financedashboard.utils.JwtUtil;
   import com.example.financedashboard.vo.UserVO;
   import org.springframework.beans.BeanUtils;
   import org.springframework.beans.factory.annotation.Autowired;
   import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
   import org.springframework.stereotype.Service;
   
   import java.time.LocalDateTime;
   
   @Service
   public class UserServiceImpl implements UserService {
       
       @Autowired
       private UserMapper userMapper;
       
       @Autowired
       private BCryptPasswordEncoder passwordEncoder;
       
       @Autowired
       private JwtUtil jwtUtil;
       
       @Override
       public boolean register(UserRegisterDTO userRegisterDTO) {
           // 检查用户名是否已存在
           if (userMapper.findByUsername(userRegisterDTO.getUsername()) != null) {
               throw new RuntimeException("用户名已存在");
           }
           
           // 检查邮箱是否已存在
           if (userMapper.findByEmail(userRegisterDTO.getEmail()) != null) {
               throw new RuntimeException("邮箱已存在");
           }
           
           // 创建用户
           User user = new User();
           BeanUtils.copyProperties(userRegisterDTO, user);
           user.setPassword(passwordEncoder.encode(userRegisterDTO.getPassword()));
           user.setStatus(1);
           user.setCreateTime(LocalDateTime.now());
           user.setUpdateTime(LocalDateTime.now());
           
           return userMapper.insert(user) > 0;
       }
       
       @Override
       public String login(UserLoginDTO userLoginDTO) {
           User user = userMapper.findByUsername(userLoginDTO.getUsername());
           if (user == null) {
               throw new RuntimeException("用户不存在");
           }
           
           if (!passwordEncoder.matches(userLoginDTO.getPassword(), user.getPassword())) {
               throw new RuntimeException("密码错误");
           }
           
           if (user.getStatus() != 1) {
               throw new RuntimeException("用户已被禁用");
           }
           
           return jwtUtil.generateToken(user);
       }
       
       @Override
       public UserVO getUserInfo(String token) {
           // 从token中获取用户信息
           String username = jwtUtil.getUsernameFromToken(token);
           User user = userMapper.findByUsername(username);
           
           if (user == null) {
               throw new RuntimeException("用户不存在");
           }
           
           UserVO userVO = new UserVO();
           BeanUtils.copyProperties(user, userVO);
           return userVO;
       }
       
       @Override
       public boolean logout(String token) {
           // 在实际应用中，可以将token加入黑名单
           return true;
       }
   }
   ```

   ```java
   // src/main/java/com/example/financedashboard/utils/JwtUtil.java
   package com.example.financedashboard.utils;
   
   import com.example.financedashboard.entity.User;
   import io.jsonwebtoken.Claims;
   import io.jsonwebtoken.Jwts;
   import io.jsonwebtoken.SignatureAlgorithm;
   import org.springframework.beans.factory.annotation.Value;
   import org.springframework.stereotype.Component;
   
   import java.util.Date;
   
   /**
    * JWT工具类
    * JWT（JSON Web Token）是一种开放标准（RFC 7519），用于在网络应用环境间传递声明的基于JSON的令牌
    * 它被设计为紧凑且安全的，特别适用于分布式站点的单点登录（SSO）场景
    * JWT由三部分组成：头部（Header）、载荷（Payload）和签名（Signature）
    */
   @Component  // Spring注解，表示该类是一个组件，会被Spring容器管理
   public class JwtUtil {
       
       @Value("${jwt.secret}")  // 从配置文件中获取JWT密钥
       private String secret;     // JWT密钥，用于签名和验证令牌
       
       @Value("${jwt.expiration}")  // 从配置文件中获取JWT过期时间
       private Long expiration;   // JWT过期时间，单位为秒
       
       /**
        * 生成JWT令牌
        * @param user 用户信息
        * @return 生成的JWT令牌字符串
        */
       public String generateToken(User user) {
           // 创建Claims对象，用于存储JWT的载荷部分
           Claims claims = Jwts.claims();
           claims.setSubject(user.getUsername());  // 设置主题，通常是用户名
           claims.put("userId", user.getId());      // 添加自定义声明，存储用户ID
           
           // 构建JWT令牌
           return Jwts.builder()
                   .setClaims(claims)                          // 设置载荷
                   .setExpiration(new Date(System.currentTimeMillis() + expiration * 1000))  // 设置过期时间
                   .signWith(SignatureAlgorithm.HS512, secret)  // 设置签名算法和密钥
                   .compact();                                 // 生成JWT字符串
       }
       
       /**
        * 从JWT令牌中获取用户名
        * @param token JWT令牌
        * @return 用户名
        */
       public String getUsernameFromToken(String token) {
           // 解析JWT令牌并获取载荷
           Claims claims = Jwts.parser()
                   .setSigningKey(secret)  // 设置签名密钥
                   .parseClaimsJws(token)  // 解析JWT令牌
                   .getBody();            // 获取载荷
           
           // 从载荷中获取主题（用户名）
           return claims.getSubject();
       }
       
       /**
        * 验证JWT令牌是否有效
        * @param token JWT令牌
        * @return 如果令牌有效返回true，否则返回false
        */
       public boolean validateToken(String token) {
           try {
               // 解析JWT令牌，如果解析过程中没有抛出异常，则令牌有效
               Jwts.parser().setSigningKey(secret).parseClaimsJws(token);
               return true;
           } catch (Exception e) {
               // 如果解析过程中抛出异常，则令牌无效
               return false;
           }
       }
   }
   ```

   ```java
   package com.example.financedashboard.interceptor;
   
   import com.example.financedashboard.utils.JwtUtil;
   import org.springframework.beans.factory.annotation.Autowired;
   import org.springframework.stereotype.Component;
   import org.springframework.web.servlet.HandlerInterceptor;
   
   import javax.servlet.http.HttpServletRequest;
   import javax.servlet.http.HttpServletResponse;
   
   @Component
   public class JwtInterceptor implements HandlerInterceptor {
       
       @Autowired
       private JwtUtil jwtUtil;
       
       @Override
       public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {
           String token = request.getHeader("Authorization");
           
           if (token != null && jwtUtil.validateToken(token)) {
               return true;
           }
           
           response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
           return false;
       }
   }
   ```

   ```java
       public boolean addStock(StockInfo stockInfo) {

           return stockInfoMapper.insert(stockInfo) > 0;
       }

       public boolean batchAddStocks(List<StockInfo> stockInfos) {
           LocalDateTime now = LocalDateTime.now();
           stockInfos.forEach(stock -> stock.setUpdateTime(now));
           return stockInfoMapper.batchInsert(stockInfos) > 0;
       }
       
       @Override
       public boolean updateStock(StockInfo stockInfo) {
           stockInfo.setUpdateTime(LocalDateTime.now());
           return stockInfoMapper.update(stockInfo) > 0;
       }
       
       @Override
       public boolean deleteStock(String stockCode) {
           return stockInfoMapper.deleteByStockCode(stockCode) > 0;
       }
       
       @Override
       public List<StockInfo> queryStocks(StockQueryDTO queryDTO) {
           // 这里可以实现更复杂的查询逻辑
           List<StockInfo> allStocks = stockInfoMapper.findAll();
           
           // 简单的过滤逻辑
           if (StringUtils.hasText(queryDTO.getKeyword())) {
               String keyword = queryDTO.getKeyword().toLowerCase();
               allStocks.removeIf(stock -> 
                   !stock.getStockCode().toLowerCase().contains(keyword) && 
                   !stock.getStockName().toLowerCase().contains(keyword));
           }
           
           if (StringUtils.hasText(queryDTO.getIndustry())) {
               allStocks.removeIf(stock -> 
                   !queryDTO.getIndustry().equals(stock.getIndustry()));
           }
           
           return allStocks;
       }
   ```

   ```java
       private String message;  // 返回消息，描述请求结果
   ```

   ```vue
           callback()
   ```

   ```java
   package com.example.financedashboard.service;
   
   import com.example.financedashboard.entity.StockInfo;
   import com.example.financedashboard.dto.IndicatorResultDTO;
   
   import java.util.List;
   import java.util.Map;
   
   /**
    * 数据分析服务接口
    * 定义了各种技术指标的计算方法和趋势分析方法
    * 技术分析是金融投资中常用的一种分析方法，通过研究历史市场数据（主要是价格和成交量）
    * 来预测未来价格走势的技术方法
    */
   public interface AnalysisService {
       /**
        * 计算移动平均线（Moving Average, MA）
        * MA是最常用的技术指标之一，通过计算一段时间内的平均价格来平滑价格数据，
        * 从而识别价格趋势。常用的MA有5日、10日、20日、30日、60日等。
        * @param stockInfos 股票信息列表
        * @param periods 移动平均线的周期，可以是多个，如5, 10, 20
        * @return 包含多个移动平均线数据的Map，键为"MA"+周期，值为移动平均线数据列表
        */
       Map<String, List<Double>> calculateMA(List<StockInfo> stockInfos, int... periods);
       
       /**
        * 计算MACD指标（Moving Average Convergence Divergence）
        * MACD称为指数平滑异同移动平均线，是从双指数移动平均线发展而来的。
        * MACD由快线（DIF）、慢线（DEA）和柱状图（MACD）三部分组成。
        * @param stockInfos 股票信息列表
        * @return 包含DIF、DEA和MACD数据的IndicatorResultDTO对象
        */
       IndicatorResultDTO calculateMACD(List<StockInfo> stockInfos);
       
       /**
        * 计算KDJ指标
        * KDJ指标又称随机指标，是一种相当新颖、实用的技术分析指标，
        * 起先用于期货市场的分析，后被广泛用于股票市场的短期趋势分析。
        * KDJ由K值、D值和J值三部分组成。
        * @param stockInfos 股票信息列表
        * @return 包含K、D和J数据的IndicatorResultDTO对象
        */
       IndicatorResultDTO calculateKDJ(List<StockInfo> stockInfos);
       
       /**
        * 计算RSI指标（Relative Strength Index）
        * RSI称为相对强弱指标，是通过比较一段时期内的平均收盘涨数和平均收盘跌数
        * 来分析市场买沽盘的意向和实力，从而作出未来市场走势的判断。
        * 常用的RSI有6日RSI、12日RSI、24日RSI等。
        * @param stockInfos 股票信息列表
        * @param periods RSI的周期，可以是多个，如6, 12, 24
        * @return 包含多个RSI数据的Map，键为"RSI"+周期，值为RSI数据列表
        */
       Map<String, List<Double>> calculateRSI(List<StockInfo> stockInfos, int... periods);
       
       /**
        * 分析价格趋势
        * 通过计算最近一段时间内价格的变化百分比，来判断价格的趋势。
        * @param stockInfos 股票信息列表

   public class AnalysisServiceImpl implements AnalysisService {
       
       /**
        * 计算移动平均线（MA）
        * 移动平均线是最常用的技术指标之一，通过计算一段时间内的平均价格来平滑价格数据，
        * 从而识别价格趋势。常用的MA有5日、10日、20日、30日、60日等。
        * @param stockInfos 股票信息列表
        * @param periods 移动平均线的周期，可以是多个，如5, 10, 20
        * @return 包含多个移动平均线数据的Map，键为"MA"+周期，值为移动平均线数据列表
        */
       @Override
       public Map<String, List<Double>> calculateMA(List<StockInfo> stockInfos, int... periods) {
           Map<String, List<Double>> result = new HashMap<>();
           
           // 遍历每个周期
           for (int period : periods) {
               List<Double> maValues = new ArrayList<>();
               
               // 遍历每个数据点
               for (int i = 0; i < stockInfos.size(); i++) {
                   // 前period-1个点没有足够的计算周期，设为null
                   if (i < period - 1) {
                       maValues.add(null);
                   } else {
                       // 计算最近period个数据点的平均值
                       double sum = 0;
                       for (int j = i - period + 1; j <= i; j++) {
                           sum += stockInfos.get(j).getCurrentPrice().doubleValue();
                       }
                       maValues.add(sum / period);
                   }
               }
               result.put("MA" + period, maValues);
           }
           
           return result;
       }
       
       /**
        * 计算MACD指标（Moving Average Convergence Divergence）
        * MACD称为指数平滑异同移动平均线，是从双指数移动平均线发展而来的。
        * MACD由快线（DIF）、慢线（DEA）和柱状图（MACD）三部分组成。
        * 计算步骤：
        * 1. 计算12日EMA（短期指数移动平均线）
        * 2. 计算26日EMA（长期指数移动平均线）
        * 3. 计算DIF = 12日EMA - 26日EMA
        * 4. 计算DEA = DIF的9日EMA
        * 5. 计算MACD柱状图 = (DIF - DEA) * 2
        * @param stockInfos 股票信息列表
        * @return 包含DIF、DEA和MACD数据的IndicatorResultDTO对象
        */
       @Override
       public IndicatorResultDTO calculateMACD(List<StockInfo> stockInfos) {
           // 数据不足26个点，无法计算MACD
           if (stockInfos.size() < 26) {
               return new IndicatorResultDTO();
           }
           
           int shortPeriod = 12; // 短期EMA周期
           int longPeriod = 26; // 长期EMA周期
           int signalPeriod = 9; // DEA周期
           
           // 提取收盘价
           List<Double> closePrices = stockInfos.stream()
                   .map(stock -> stock.getCurrentPrice().doubleValue())
                   .toList();
           
           // 计算短期EMA和长期EMA
           List<Double> shortEMA = calculateEMA(closePrices, shortPeriod);
           List<Double> longEMA = calculateEMA(closePrices, longPeriod);
           
           // 计算DIF（差离值）
           List<Double> dif = new ArrayList<>();
           for (int i = 0; i < shortEMA.size(); i++) {
               dif.add(shortEMA.get(i) - longEMA.get(i));
           }
           
           // 计算DEA（信号线）
           List<Double> dea = calculateEMA(dif, signalPeriod);
           
           // 计算MACD柱状图
           List<Double> macd = new ArrayList<>();
           for (int i = 0; i < dif.size(); i++) {
               macd.add((dif.get(i) - dea.get(i)) * 2);
           }
           
           // 封装结果
           IndicatorResultDTO result = new IndicatorResultDTO();
           result.setDif(dif);
           result.setDea(dea);
           result.setMacd(macd);
           
           return result;
       }
       
       /**
        * 计算KDJ指标
        * KDJ指标又称随机指标，是一种相当新颖、实用的技术分析指标，
        * 起先用于期货市场的分析，后被广泛用于股票市场的短期趋势分析。
        * KDJ由K值、D值和J值三部分组成。
        * 计算步骤：
        * 1. 计算RSV（未成熟随机值）= (当日收盘价 - N日内最低价) / (N日内最高价 - N日内最低价) * 100
        * 2. 计算K值 = 当日K值 = (1/3) * 当日RSV + (2/3) * 前一日K值
        * 3. 计算D值 = 当日D值 = (1/3) * 当日K值 + (2/3) * 前一日D值
        * 4. 计算J值 = 3 * 当日K值 - 2 * 当日D值
        * @param stockInfos 股票信息列表
        * @return 包含K、D和J数据的IndicatorResultDTO对象
        */
       @Override
       public IndicatorResultDTO calculateKDJ(List<StockInfo> stockInfos) {
           // 数据不足9个点，无法计算KDJ
           if (stockInfos.size() < 9) {
               return new IndicatorResultDTO();
           }
           
           int period = 9; // KDJ周期
           
           // 提取最高价、最低价和收盘价
           List<Double> highPrices = stockInfos.stream()
                   .map(stock -> stock.getHighPrice().doubleValue())
                   .toList();
           
           List<Double> lowPrices = stockInfos.stream()
                   .map(stock -> stock.getLowPrice().doubleValue())
                   .toList();
           
           List<Double> closePrices = stockInfos.stream()
                   .map(stock -> stock.getCurrentPrice().doubleValue())
                   .toList();
           
           List<Double> rsv = new ArrayList<>();
           List<Double> k = new ArrayList<>();
           List<Double> d = new ArrayList<>();
           List<Double> j = new ArrayList<>();
           
           // 计算RSV（未成熟随机值）
           for (int i = period - 1; i < stockInfos.size(); i++) {
               double high = Collections.max(highPrices.subList(i - period + 1, i + 1));
               double low = Collections.min(lowPrices.subList(i - period + 1, i + 1));
               double rsvValue = (closePrices.get(i) - low) / (high - low) * 100;
               rsv.add(rsvValue);
           }
           
           // 计算K值
           k.add(50.0); // 第一个K值设为50
           for (int i = 1; i < rsv.size(); i++) {
               double kValue = (k.get(i - 1) * 2 + rsv.get(i)) / 3;
               k.add(kValue);
           }
           
           // 计算D值
           d.add(50.0); // 第一个D值设为50
           for (int i = 1; i < k.size(); i++) {
               double dValue = (d.get(i - 1) * 2 + k.get(i)) / 3;
               d.add(dValue);
           }
           
           // 计算J值
           for (int i = 0; i < k.size(); i++) {
               double jValue = 3 * k.get(i) - 2 * d.get(i);
               j.add(jValue);
           }
           
           // 填充前面的空值，使数据长度与原始数据一致
           List<Double> fullK = new ArrayList<>(Collections.nCopies(period - 1, null));
           fullK.addAll(k);
           
           List<Double> fullD = new ArrayList<>(Collections.nCopies(period - 1, null));
           fullD.addAll(d);
           
           List<Double> fullJ = new ArrayList<>(Collections.nCopies(period - 1, null));
           fullJ.addAll(j);
           
           // 封装结果
           IndicatorResultDTO result = new IndicatorResultDTO();
           result.setK(fullK);
           result.setD(fullD);
           result.setJ(fullJ);
           
           return result;
       }
       
       /**
        * 计算RSI指标（Relative Strength Index）
        * RSI称为相对强弱指标，是通过比较一段时期内的平均收盘涨数和平均收盘跌数
        * 来分析市场买沽盘的意向和实力，从而作出未来市场走势的判断。
        * 计算步骤：
        * 1. 计算每个交易日的价格变化 = 当日收盘价 - 前一日收盘价
        * 2. 计算N日内的平均上涨幅度和平均下跌幅度
        * 3. 计算RS（相对强度）= 平均上涨幅度 / 平均下跌幅度
        * 4. 计算RSI = 100 - (100 / (1 + RS))
        * @param stockInfos 股票信息列表
        * @param periods RSI的周期，可以是多个，如6, 12, 24
        * @return 包含多个RSI数据的Map，键为"RSI"+周期，值为RSI数据列表
        */
       @Override
       public Map<String, List<Double>> calculateRSI(List<StockInfo> stockInfos, int... periods) {
           Map<String, List<Double>> result = new HashMap<>();
           
           // 提取收盘价
           List<Double> closePrices = stockInfos.stream()
                   .map(stock -> stock.getCurrentPrice().doubleValue())
                   .toList();
           
           // 遍历每个周期
           for (int period : periods) {
               List<Double> rsiValues = new ArrayList<>();
               
               // 计算价格变化
               List<Double> changes = new ArrayList<>();
               for (int i = 1; i < closePrices.size(); i++) {
                   changes.add(closePrices.get(i) - closePrices.get(i - 1));
               }
               
               // 计算RSI
               for (int i = 0; i < changes.size(); i++) {
                   // 前period-1个点没有足够的计算周期，设为null
                   if (i < period - 1) {
                       rsiValues.add(null);
                   } else {
                       double gain = 0;
                       double loss = 0;
                       
                       // 计算N日内的总上涨幅度和总下跌幅度
                       for (int j = i - period + 1; j <= i; j++) {
                           if (changes.get(j) > 0) {
                               gain += changes.get(j);
                           } else {
                               loss -= changes.get(j);
                           }
                       }
                       
                       // 计算平均上涨幅度和平均下跌幅度
                       double avgGain = gain / period;
                       double avgLoss = loss / period;
                       
                       // 计算RS（相对强度）
                       double rs = avgGain / avgLoss;
                       
                       // 计算RSI
                       double rsi = 100 - (100 / (1 + rs));
                       
                       rsiValues.add(rsi);
                   }
               }
               
               // 在前面添加一个空值，使RSI列表与原始数据长度一致
               rsiValues.add(0, null);
               result.put("RSI" + period, rsiValues);
           }
           
           return result;
       }
       
       /**
        * 分析价格趋势
        * 通过计算最近一段时间内价格的变化百分比，来判断价格的趋势。
        * @param stockInfos 股票信息列表
        * @return 趋势描述字符串，如"强势上涨"、"温和上涨"、"横盘整理"等
        */
       @Override
       public String analyzePriceTrend(List<StockInfo> stockInfos) {
           // 数据不足5个点，无法分析趋势
           if (stockInfos.size() < 5) {
               return "数据不足，无法分析趋势";
           }
           
           // 获取最近5天的数据
           List<StockInfo> recentData = stockInfos.subList(Math.max(0, stockInfos.size() - 5), stockInfos.size());
           
           // 计算价格变化百分比
           double firstPrice = recentData.get(0).getCurrentPrice().doubleValue();
           double lastPrice = recentData.get(recentData.size() - 1).getCurrentPrice().doubleValue();
           double changePercent = (lastPrice - firstPrice) / firstPrice * 100;
           
           // 根据变化百分比判断趋势
           if (changePercent > 5) {
               return "强势上涨";
           } else if (changePercent > 2) {
               return "温和上涨";
           } else if (changePercent > -2) {
               return "横盘整理";
           } else if (changePercent > -5) {
               return "温和下跌";
           } else {
               return "强势下跌";
           }
       }
       
       /**
        * 分析成交量趋势
        * 通过计算最近一段时间内成交量的变化百分比，来判断成交量的趋势。
        * @param stockInfos 股票信息列表
        * @return 趋势描述字符串，如"成交量大幅增加"、"成交量温和增加"等
        */
       @Override
       public String analyzeVolumeTrend(List<StockInfo> stockInfos) {
           // 数据不足5个点，无法分析趋势
           if (stockInfos.size() < 5) {
               return "数据不足，无法分析趋势";
           }
           
           // 获取最近5天的数据
           List<StockInfo> recentData = stockInfos.subList(Math.max(0, stockInfos.size() - 5), stockInfos.size());
           
           // 计算成交量变化百分比
           long firstVolume = recentData.get(0).getVolume();
           long lastVolume = recentData.get(recentData.size() - 1).getVolume();
           double changePercent = (double) (lastVolume - firstVolume) / firstVolume * 100;
           
           // 根据变化百分比判断趋势
           if (changePercent > 50) {
               return "成交量大幅增加";
           } else if (changePercent > 20) {
               return "成交量温和增加";
           } else if (changePercent > -20) {
               return "成交量稳定";
           } else if (changePercent > -50) {
               return "成交量温和减少";
           } else {
               return "成交量大幅减少";
           }
       }
       
       /**
        * 计算EMA（指数移动平均线）
        * EMA（Exponential Moving Average）是一种给予近期数据更高权重的移动平均线，
        * 相比简单移动平均线（SMA），EMA对价格变化更敏感。
        * 计算公式：EMA = 当日收盘价 * 平滑系数 + 前一日EMA * (1 - 平滑系数)
        * 其中，平滑系数 = 2 / (N + 1)，N为周期
        * @param data 数据列表
        * @param period EMA周期
        * @return EMA值列表
        */
       private List<Double> calculateEMA(List<Double> data, int period) {
           List<Double> ema = new ArrayList<>();
           
           // 第一个EMA值就是第一个数据点
           ema.add(data.get(0));
           
           // 计算平滑系数
           double multiplier = 2.0 / (period + 1);
           
           // 计算后续的EMA值
           for (int i = 1; i < data.size(); i++) {
               double emaValue = data.get(i) * multiplier + ema.get(i - 1) * (1 - multiplier);
               ema.add(emaValue);
           }
           
           return ema;
       }
   }
   ```

   ```vue
       rsiForm.periods.forEach(period => {
         const rsiValues = []
         for (let i = 0; i < dates.length; i++) {
           rsiValues.push(Math.random() * 100)
         }
         series.push({
           name: 'RSI' + period,
           type: 'line',
           data: rsiValues
         })
       })
       
       return {
         xAxis: dates,
         legend: rsiForm.periods.map(p => 'RSI' + p),
         series: series
       }
     }
   
     // ... 其他代码保持不变 ...
   }
   </script>
   ```
